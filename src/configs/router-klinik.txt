!
! Router-Klinik — Cisco ISR4331
! Penghubung jaringan klinik ke internet.
!
! Alamat mengikuti docs/pengalamatan-vlsm.md:
!   Gig0/0/1 ke Core-Switch-L3 : 172.16.1.1/30
!   Gig0/0/0 ke internet        : dari penyedia (pada simulasi 10.0.0.2/30)
!
! Salinan ini disalin apa adanya dari perangkat simulasi setelah diuji.
! Nomor antarmuka wajib dicek ulang bila topologi Packet Tracer diubah.
!
enable
configure terminal
 hostname Router-Klinik

 ! --- ke core switch ---
 interface GigabitEthernet0/0/1
  description Tautan ke Core-Switch-L3
  ip address 172.16.1.1 255.255.255.252
  no shutdown
 exit

 ! --- ke internet (alamat dari penyedia) ---
 interface GigabitEthernet0/0/0
  description Tautan ke internet
  ip address 10.0.0.2 255.255.255.252
  ip nat outside
  no shutdown
 exit

 ! --- rute bawaan ke internet ---
 ip route 0.0.0.0 0.0.0.0 10.0.0.1

 ! --- NAT overload: alamat privat keluar memakai alamat antarmuka luar ---
 ! Catatan: ACL ini menentukan siapa yang boleh keluar lewat NAT. Pemisahan
 ! tamu dari jaringan medis TIDAK dilakukan di sini, melainkan oleh ACL
 ! BATAS_TAMU pada Core-Switch-L3 — jadi VLAN 99 tetap diizinkan keluar ke
 ! internet (sesuai README dan docs/konfigurasi.md), sementara aksesnya ke
 ! VLAN internal sudah ditutup lebih dulu di core switch.
 ip access-list standard DAPAT_KELUAR
  permit 192.168.10.0 0.0.0.63     ! VLAN 10, 20 (pendaftaran, poliklinik)
  permit 192.168.10.128 0.0.0.31   ! VLAN 30, 40 (farmasi, laboratorium)
  permit 192.168.10.192 0.0.0.15   ! VLAN 50 (manajemen)
  permit 192.168.10.208 0.0.0.15   ! VLAN 99 (tamu) — internet saja, ke VLAN internal sudah ditolak BATAS_TAMU
  permit 192.168.10.240 0.0.0.7    ! VLAN 100 (server) — pembaruan sistem dan DNS luar
  deny   any
 exit
 ip nat inside source list DAPAT_KELUAR interface GigabitEthernet0/0/0 overload

 ! --- tandai sisi dalam NAT ---
 interface GigabitEthernet0/0/1
  ip nat inside
 exit

 ! --- routing dinamis bersama core switch ---
 router ospf 1
  router-id 1.1.1.1
  network 172.16.1.0 0.0.0.3 area 0
  default-information originate
 exit

 ! --- batasi akses manajemen dari luar ---
 line vty 0 4
  transport input ssh
  login local
 exit
 username admin privilege 15 secret GANTI_SANDI_INI
 ip domain-name klinik.local
 crypto key generate rsa modulus 2048

! --- simpan ---
end
write memory
